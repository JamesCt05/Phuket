    function switchDay(day) {

      document.querySelectorAll('.day-tab').forEach((tab, index) => {

        if (index + 1 === day) {

          tab.className = "day-tab px-4 py-2 rounded-xl text-sm font-semibold transition bg-teal-600 text-white shrink-0 shadow-sm";

        } else {

          tab.className = "day-tab px-4 py-2 rounded-xl text-sm font-semibold transition bg-slate-100 text-slate-600 hover:bg-slate-200 shrink-0";

        }

      });

      renderDay(day);

    }
